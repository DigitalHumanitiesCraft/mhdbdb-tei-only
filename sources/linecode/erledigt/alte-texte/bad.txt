0000000000000000001 VON DER ÜBELN ADELHEIT
0000000000000000002 UND IREM MAN
0000000000000000010 In einem dorf was gesezzen ein man,
0000000000000000020 %der nie kein guoten tac gewan.%
0000000000000000030 daz geschach von sînem wîbe:
0000000000000000040 si swuor bî irem lîbe,
0000000000000000050 daz si nimmer wolt werden guot.
0000000000000000060 daz beswert im sînen muot.
0000000000000000070 er was geheizen der guot Markhart
0000000000000000080 - wê im, daz er ie geborn wart! -
0000000000000000090 und sî diu übel Adelheit.
0000000000000000100 si tet im %jârlanc% leit.
0000000000000000110 Eins tages sâzens ob dem viure;
0000000000000000120 der imbiz was im tiure.
0000000000000000130 er sprach: "liebe Adelheit,
0000000000000000140 ist der imbiz iht schier bereit?
0000000000000000150 gip uns zessen, daz dir got lône;
0000000000000000160 ez nâhet schier diu zît der nône."
0000000000000000170 "und wær ez dîn grimmiger tôt,
0000000000000000180 du enbîzest tâlanc kein gebrôt,
0000000000000000190 du muost noch hiut vasten,
0000000000000000200 biz dir dîn ougen glasten."
0000000000000000210 guot Markhart het ein pfenninc,
0000000000000000220 dâmit schafft er sîn dinc.
0000000000000000230 er wolt in daz dorf loufen
0000000000000000240 und im ein brôt koufen.
0000000000000000250 dô muost er sich ê roufen,
0000000000000000260 mit dem übeln wîb boufen.
0000000000000000270 si sluoc in und stiez,
0000000000000000280 daz %im% nieman gehiez
0000000000000000290 sîn leben vür den tôt,
0000000000000000300 hæt er genomen daz brôt.
0000000000000000310 er gedâht in sînem sinne:
0000000000000000320 "wes sol ich beginnen,
0000000000000000330 daz ich die vâlentinne
0000000000000000340 ûf mînen weg bringe?"
0000000000000000350 Dô er sîn nôt überwant,
0000000000000000360 der guot Markhart gienc zuohant
0000000000000000370 hin under sîn selbes tür.
0000000000000000380 dâ gienc manec man hin vür,
0000000000000000390 die gen Auspurc wolten gân.
0000000000000000400 daz merk vrouwe und man:
0000000000000000410 der guot Markhart moht niht lân,
0000000000000000420 er huop von ezzen wider an.
0000000000000000430 er sprach: "liebe Adelheit
0000000000000000440 - daz dir geschehe nimmer leit! -
0000000000000000450 sich, dâ gât manec man hin,
0000000000000000460 mich trieg dann mîn sin,
0000000000000000470 belib er %heim%, ez dûht mich guot.
0000000000000000480 er vertrinkt hiut mantel und %huot%."
0000000000000000490 si sprach: "du wirst niht erlôn,
0000000000000000500 du muost ouch zuo markt gon."
0000000000000000510 er sprach: "liebe Adelheit
0000000000000000520 - dir geschehe %nimmer% leit! -
0000000000000000530 daz best ich dir râten sol:
0000000000000000540 belîp hie heim und hüet wol!"
0000000000000000550 si sprach: "hab dir den rât,
0000000000000000560 wan durch dich nieman lât!
0000000000000000570 kanstu mich verstân?
0000000000000000580 ich wil ouch zuo markt gân."
0000000000000000590 er sprach: "merkz, wie ichz meine:
0000000000000000600 lâ die pfenninge hie heime,
0000000000000000610 darumb ich %gap mîn% guot rint!
0000000000000000620 des hânt schaden mîniu kint."
0000000000000000630 si sprach: "ich wil dir sagen:
0000000000000000640 ich wil si selbs mit mir tragen
0000000000000000650 und wil dâvon zern;
0000000000000000660 daz kan mir nieman %entwern%.."
0000000000000000670 Diu wîle wert unlange,
0000000000000000680 dô kam ein man gegangen,
0000000000000000690 den het ein rock umbvangen
0000000000000000700 (der selbe rock, der was blâ)
0000000000000000710 und ein zwîfachen schaprûn grâ.
0000000000000000720 er truoc ein guot swert
0000000000000000730 und einen niuwen huot wert.
0000000000000000740 der guot Markhart gemeit
0000000000000000750 sprach zuo sîner Adelheit:
0000000000000000760 "nu luoc zuo disem affen!
0000000000000000770 wie ist er geschaffen!
0000000000000000780 er wirt ûz im machen,
0000000000000000790 daz man sîn beginnet lachen.
0000000000000000800 er tregt einen blâwen rock,
0000000000000000810 man wirt in ankaffen %als ein bock%;
0000000000000000820 ein swert tregt er und einen huot.
0000000000000000830 ez dunkt mich niht guot."
0000000000000000840 si sprach: "du wirst es niht erlân,
0000000000000000850 du muost ouch ein blâwen rock hân."
0000000000000000860 er sprach: "guot Adelheit,
0000000000000000870 als liep ich dir sî geseit,
0000000000000000880 des erlâz mich durch got,
0000000000000000890 wan ich würd der liute spot!"
0000000000000000900 si sprach: "du wirst es niht erlân!
0000000000000000910 du muost ein blâwen rock hân."
0000000000000000920 er sprach: "liebe Adelheit
0000000000000000930 - dir geschehe %nimmer% leit! -
0000000000000000940 so kouf mir des bresten!"
0000000000000000950 si sprach: "%nein%, des besten,
0000000000000000960 des ich zuo Auspurc vinde
0000000000000000970 veil umb mîn pfenninge."
0000000000000000980 Diu wîle wert unlange;
0000000000000000990 si kâmen in die stat gangen.

0000000000000001010 so siz an dem markte mohte weln,
0000000000000001020 und hiez daz snîden schier
0000000000000001030 umb guoter pfennige vier
0000000000000001040 und hiez in machen wol,
0000000000000001050 als in ein biderman tragen sol.
0000000000000001060 dô der rock was bereit,

0000000000000001140 wîn ich niht trinken sol,
0000000000000001150 wazzer tuot mir alsô wol."
0000000000000001160 si sprach: "wær ez dîn grimmer tôt,
0000000000000001170 du muost ezzen weizbrôt
0000000000000001180 und trinken den besten wîn,
0000000000000001190 sô er iendert hie mac gesîn."
0000000000000001200 dâmit wolt si in toeten;
0000000000000001210 si wolt in vröuden noeten.
0000000000000001220 si vuort in zuohant

0000000000000001240 dâ selbst si nider sâzen,
0000000000000001250 trunken unde âzen,
0000000000000001260 daz ir dinc wol stuont,
0000000000000001270 als noch vil liute tuont.

0000000000000001290 - dir geschehe %nimmer% leit -
0000000000000001300 merk, waz ich dir sage:
0000000000000001310 wir suln trinken %un z ze% tage!"
0000000000000001320 Dô Adelheit die rede vernam,
0000000000000001330 bald huop si sich von dan.
0000000000000001340 Adelheit diu vil vreche
0000000000000001350 lief ze tal bîm Leche.
0000000000000001360 si trat ze %nâch% ûf daz gestat

0000000000000001380 er sprach: "%tritâ% herdan baz!"
0000000000000001390 si sprach: "warumbe tæt ich daz?
0000000000000001400 %sît du mich sîn hâst gebeten%,
0000000000000001410 ich wil hinzuo baz treten!"
0000000000000001420 daz schuof ir unreiner haz,
0000000000000001430 daz si hinîn blatzte baz.
0000000000000001440 der Lech truoc si an der stunt
0000000000000001450 vast an den tiefen grunt,
0000000000000001460 daz er si niemer mêr gesach.

0000000000000001810 und gesuocht sie nimmer mêre.
